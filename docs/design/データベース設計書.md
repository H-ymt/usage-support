## 2. テーブル定義

### 2.1 users（ユーザー）

LINE認証したユーザーの基本情報を保持する。

```sql
CREATE TABLE users (
    user_key VARCHAR(255) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    email VARCHAR(255),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE users IS 'LINEミニアプリユーザー';
COMMENT ON COLUMN users.user_key IS 'LINE LIFF user_key（チャネル固有の識別子）';
COMMENT ON COLUMN users.name IS '氏名';
COMMENT ON COLUMN users.phone IS '電話番号';
COMMENT ON COLUMN users.email IS 'メールアドレス（任意）';
```

---

### 2.2 stores（店舗）

店舗マスタ。

```sql
CREATE TABLE stores (
    store_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    address VARCHAR(255),
    phone VARCHAR(20),
    business_hours VARCHAR(100),
    is_active BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE stores IS '店舗マスタ';
COMMENT ON COLUMN stores.is_active IS '有効フラグ（false: 非表示）';
```

---

### 2.3 loan_tokens（貸出開始用トークン）

店頭QRコードに埋め込むトークンを管理する。

```sql
CREATE TABLE loan_tokens (
    token_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    store_id UUID NOT NULL REFERENCES stores(store_id),
    token VARCHAR(255) NOT NULL UNIQUE,
    expires_at TIMESTAMPTZ NOT NULL,
    used_at TIMESTAMPTZ,
    used_by VARCHAR(255) REFERENCES users(user_key),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE loan_tokens IS '貸出開始登録用QRトークン';
COMMENT ON COLUMN loan_tokens.token IS 'QRコードに埋め込むトークン文字列';
COMMENT ON COLUMN loan_tokens.expires_at IS 'トークン有効期限';
COMMENT ON COLUMN loan_tokens.used_at IS '使用日時（NULLなら未使用）';
COMMENT ON COLUMN loan_tokens.used_by IS '使用したユーザー';
```

### 2.4 loans（貸出）

90日間の貸出情報を管理する。

```sql
CREATE TYPE loan_status AS ENUM ('active', 'ended');

CREATE TABLE loans (
    loan_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_key VARCHAR(255) NOT NULL REFERENCES users(user_key),
    store_id UUID NOT NULL REFERENCES stores(store_id),
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    started_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    ended_at TIMESTAMPTZ,
    status loan_status NOT NULL DEFAULT 'active',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chk_end_date CHECK (end_date = start_date + INTERVAL '90 days')
);

COMMENT ON TABLE loans IS '補聴器貸出';
COMMENT ON COLUMN loans.start_date IS '貸出開始日（登録当日）';
COMMENT ON COLUMN loans.end_date IS '貸出終了日（開始日+90日）';
COMMENT ON COLUMN loans.started_at IS '開始登録日時';
COMMENT ON COLUMN loans.ended_at IS '終了処理日時';
COMMENT ON COLUMN loans.status IS 'ステータス（active: 進行中, ended: 終了）';
```

### 2.5 surveys（アンケート）

30/60/90日アンケートの回答を保持する。

```sql
CREATE TYPE survey_type AS ENUM ('30', '60', '90');

CREATE TABLE surveys (
    survey_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    loan_id UUID NOT NULL REFERENCES loans(loan_id),
    type survey_type NOT NULL,
    answered_at TIMESTAMPTZ,
    answers JSONB,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_loan_survey_type UNIQUE (loan_id, type)
);

COMMENT ON TABLE surveys IS '使用状況アンケート';
COMMENT ON COLUMN surveys.type IS 'アンケート種別（30/60/90日）';
COMMENT ON COLUMN surveys.answered_at IS '回答日時（NULLなら未回答）';
COMMENT ON COLUMN surveys.answers IS '回答内容（JSON形式）';
```

**answers JSONの構造例（30日）:**

```json
{
  "q1_daily_use": "yes",
  "q2_hours": "8_plus",
  "q3_satisfaction": 4,
  "q4_discomfort": "none",
  "q5_concerns": "特になし"
}
```

### 2.6 reservations（予約）

来店予約を管理する。

```sql
CREATE TYPE reservation_status AS ENUM ('confirmed', 'cancelled', 'completed');

CREATE TABLE reservations (
    reservation_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    loan_id UUID NOT NULL REFERENCES loans(loan_id),
    store_id UUID NOT NULL REFERENCES stores(store_id),
    reserved_at TIMESTAMPTZ NOT NULL,
    status reservation_status NOT NULL DEFAULT 'confirmed',
    cancelled_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE reservations IS '来店予約';
COMMENT ON COLUMN reservations.reserved_at IS '予約日時';
COMMENT ON COLUMN reservations.status IS 'ステータス（confirmed: 確定, cancelled: キャンセル, completed: 完了）';
```

### 2.7 admin_users（管理者）

管理画面の認証情報はSupabase Authで管理し、このテーブルは追加属性を保持する。

```sql
CREATE TYPE admin_role AS ENUM ('admin', 'staff');

CREATE TABLE admin_users (
    admin_id UUID PRIMARY KEY REFERENCES auth.users(id),
    email VARCHAR(255) NOT NULL UNIQUE,
    name VARCHAR(100) NOT NULL,
    role admin_role NOT NULL DEFAULT 'staff',
    store_id UUID REFERENCES stores(store_id),
    is_active BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE admin_users IS '管理者';
COMMENT ON COLUMN admin_users.admin_id IS 'Supabase Auth user ID';
COMMENT ON COLUMN admin_users.role IS '権限（admin: 本部管理者, staff: 店舗スタッフ）';
COMMENT ON COLUMN admin_users.store_id IS '所属店舗（staffの場合必須）';
```

## 3. インデックス

```sql
-- users
CREATE INDEX idx_users_phone ON users(phone);
CREATE INDEX idx_users_email ON users(email) WHERE email IS NOT NULL;

-- loan_tokens
CREATE INDEX idx_loan_tokens_store ON loan_tokens(store_id);
CREATE INDEX idx_loan_tokens_expires ON loan_tokens(expires_at) WHERE used_at IS NULL;

-- loans
CREATE INDEX idx_loans_user ON loans(user_key);
CREATE INDEX idx_loans_store ON loans(store_id);
CREATE INDEX idx_loans_status ON loans(status) WHERE status = 'active';
CREATE INDEX idx_loans_end_date ON loans(end_date) WHERE status = 'active';

-- surveys
CREATE INDEX idx_surveys_loan ON surveys(loan_id);
CREATE INDEX idx_surveys_unanswered ON surveys(loan_id) WHERE answered_at IS NULL;

-- reservations
CREATE INDEX idx_reservations_loan ON reservations(loan_id);
CREATE INDEX idx_reservations_store ON reservations(store_id);
CREATE INDEX idx_reservations_date ON reservations(reserved_at) WHERE status = 'confirmed';

-- admin_users
CREATE INDEX idx_admin_users_store ON admin_users(store_id) WHERE store_id IS NOT NULL;
```

## 4. RLSポリシー

### 4.1 ユーザー向けテーブル（LINEミニアプリ経由）

ユーザーは自分のデータのみアクセス可能。

```sql
-- users
ALTER TABLE users ENABLE ROW LEVEL SECURITY;

CREATE POLICY users_select ON users
    FOR SELECT USING (user_key = current_setting('app.user_key', true));

CREATE POLICY users_insert ON users
    FOR INSERT WITH CHECK (user_key = current_setting('app.user_key', true));

CREATE POLICY users_update ON users
    FOR UPDATE USING (user_key = current_setting('app.user_key', true));

-- loans
ALTER TABLE loans ENABLE ROW LEVEL SECURITY;

CREATE POLICY loans_select ON loans
    FOR SELECT USING (user_key = current_setting('app.user_key', true));

CREATE POLICY loans_insert ON loans
    FOR INSERT WITH CHECK (user_key = current_setting('app.user_key', true));

-- surveys（loan経由でユーザーを特定）
ALTER TABLE surveys ENABLE ROW LEVEL SECURITY;

CREATE POLICY surveys_select ON surveys
    FOR SELECT USING (
        loan_id IN (
            SELECT loan_id FROM loans
            WHERE user_key = current_setting('app.user_key', true)
        )
    );

CREATE POLICY surveys_update ON surveys
    FOR UPDATE USING (
        loan_id IN (
            SELECT loan_id FROM loans
            WHERE user_key = current_setting('app.user_key', true)
        )
    );

-- reservations（loan経由でユーザーを特定）
ALTER TABLE reservations ENABLE ROW LEVEL SECURITY;

CREATE POLICY reservations_select ON reservations
    FOR SELECT USING (
        loan_id IN (
            SELECT loan_id FROM loans
            WHERE user_key = current_setting('app.user_key', true)
        )
    );

CREATE POLICY reservations_insert ON reservations
    FOR INSERT WITH CHECK (
        loan_id IN (
            SELECT loan_id FROM loans
            WHERE user_key = current_setting('app.user_key', true)
        )
    );

CREATE POLICY reservations_update ON reservations
    FOR UPDATE USING (
        loan_id IN (
            SELECT loan_id FROM loans
            WHERE user_key = current_setting('app.user_key', true)
        )
    );
```

### 4.2 公開テーブル（読み取り専用）

```sql
-- stores（有効な店舗のみ全ユーザーが参照可能）
```

## 5. マイグレーション

### 5.1 初期マイグレーション

```sql
-- 000_initial.sql

-- 拡張機能
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ENUMタイプ
CREATE TYPE loan_status AS ENUM ('active', 'ended');
CREATE TYPE survey_type AS ENUM ('30', '60', '90');
CREATE TYPE reservation_status AS ENUM ('confirmed', 'cancelled', 'completed');
CREATE TYPE admin_role AS ENUM ('admin', 'staff');

-- テーブル作成（上記定義を順番に実行）

-- updated_at自動更新用トリガー
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_users_updated_at
    BEFORE UPDATE ON users
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER trg_stores_updated_at
    BEFORE UPDATE ON stores
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER trg_loans_updated_at
    BEFORE UPDATE ON loans
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER trg_reservations_updated_at
    BEFORE UPDATE ON reservations
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER trg_admin_users_updated_at
    BEFORE UPDATE ON admin_users
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();
```

## 補足

### タイムゾーン処理

- すべてのTIMESTAMP型は`TIMESTAMPTZ`（タイムゾーン付き）を使用
- アプリケーション側でJSTに変換して表示
- DATE型はタイムゾーンなし（日付のみ）

### 削除方針

- 物理削除は行わない
- 論理削除が必要な場合は`is_active`フラグまたは`deleted_at`カラムを追加
